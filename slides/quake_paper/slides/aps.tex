  \2 Uniform Density in query Hypersphere
  \2 Partition = half space
  \2 Partition intersection is sphere cap

$$
\text{Recall}_i = \frac{\text{Vol}\left(B(q, \rho_k) \cap P_i\right)}{\text{Vol}\left(B(q, \rho_k) \right)}
$$

\centering

  p_i &= (1 - p_0)v_i
